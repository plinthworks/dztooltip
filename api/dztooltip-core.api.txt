## PopoverCloseReason
  export type PopoverCloseReason = 'escape' | 'outside' | 'focus' | 'trigger' | 'code' | 'group' | 'gone' | 'leave';
## PopoverHandle
  export interface PopoverHandle {
      open(): void;
      close(): void;
      toggle(): void;
      readonly isOpen: boolean;
      readonly el: HTMLElement | null;
      reposition(): void;
      update(o: Partial<PopoverOptions>): void;
      destroy(): void;
  }
## PopoverOptions
  export interface PopoverOptions extends TipLook {
      readonly content: TipContent;
      readonly label?: string;
      readonly group?: string;
      readonly openOn?: 'click' | 'hover' | 'manual';
      readonly delayMs?: number;
      readonly hideDelayMs?: number;
      readonly modal?: boolean;
      readonly initialFocus?: HTMLElement | ((bubble: HTMLElement) => HTMLElement | null | undefined) | false;
      readonly returnFocus?: boolean | HTMLElement;
      readonly nonce?: string;
      readonly root?: HTMLElement;
      readonly onOpen?: () => void;
      readonly onClose?: (reason: PopoverCloseReason) => void;
  }

  export interface TipAnimateInfo {

## TipAnimation
  export type TipAnimation = 'fade' | 'scale' | 'slide' | 'pop' | 'rise' | 'none' | 'morph';
## TipContent
  export type TipContent = string | Node | {
      readonly html: string;
  } | (() => string | Node | {
      readonly node: Node;
      readonly gone?: () => void;
  });
## TipLook
  export interface TipLook {
      readonly tone?: TipTone;
      readonly animation?: TipAnimation;
      readonly animate?: TipAnimate;
      readonly reducedMotion?: 'none' | 'fade';
      readonly placement?: TooltipPlacement;
      readonly flip?: boolean;
      readonly shape?: string;
      readonly style?: Readonly<Record<`--${string}`, string>>;
      readonly maxWidth?: number | string;
      readonly offset?: number | TipOffset | ((s: TipOffsetState) => number | TipOffset);
      readonly shift?: boolean | 'attached';
      readonly everyFrame?: boolean;
  }
## TipOffset
  export interface TipOffset {
      readonly mainAxis?: number;
      readonly crossAxis?: number;
      readonly alignmentAxis?: number | null;
  }
## TipOffsetState
  export interface TipOffsetState {
      readonly side: TipSide;
      readonly align: TipAlign;
      readonly anchor: {
          readonly inlineSize: number;
          readonly blockSize: number;
      };
      readonly bubble: {
          readonly inlineSize: number;
          readonly blockSize: number;
      };
  }
## TipSide
  export type TipSide = 'block-end' | 'block-start' | 'inline-end' | 'inline-start';
## TipThenable
  export interface TipThenable {
      then(onDone: (value: unknown) => unknown): unknown;
  }
## TipTone
  export type TipTone = 'info' | 'neutral' | 'success' | 'warning' | 'danger';
## TooltipHandle
  export interface TooltipHandle {
      show(): void;
      hide(): void;
      update(o: Partial<TooltipOptions>): void;
      readonly shown: boolean;
      destroy(): void;
  }
## TooltipOptions
  export interface TooltipOptions extends TipLook {
      readonly content: TipContent;
      readonly delayMs?: number;
      readonly hideDelayMs?: number;
      readonly disabled?: boolean;
      readonly when?: 'always' | 'truncated';
      readonly nonce?: string;
      readonly group?: string | false;
      readonly root?: HTMLElement;
      readonly followCursor?: boolean | 'inline' | 'block';
  }
